function update(dt) {
  if (state !== 'playing') return;

  // Update player angle
  player.angle += ANGULAR_SPEED * dt;
  // keep angle within 0..2π
  if (player.angle > Math.PI * 2) player.angle -= Math.PI * 2;

  // Update radial velocity and position
  player.vr += GRAVITY_ACCEL * gravityDirection * dt;
  player.radius += player.vr * dt;

  // Clamp radius within tunnel
  if (player.radius < RADIUS_INNER) {
    player.radius = RADIUS_INNER;
    player.vr = 0;
  } else if (player.radius > RADIUS_OUTER) {
    player.radius = RADIUS_OUTER;
    player.vr = 0;
  }

  // Spawn hazards
  spawnTimer -= dt;
  if (spawnTimer <= 0) {
    spawnTimer = 0.6 + Math.random() * 0.8; // 0.6-1.4 sec
    const lane = Math.random() < 0.5 ? 'inner' : 'outer';
    hazards.push({ angle: Math.PI * 2, lane });
  }

  // Update hazards
  for (let i = hazards.length - 1; i >= 0; i--) {
    const h = hazards[i];
    h.angle -= ANGULAR_SPEED * dt;
    if (h.angle <= 0) {
      // hazard reached player
      const playerLane = player.radius < LANE_MID ? 'inner' : 'outer';
      if (h.lane === playerLane) {
        // collision
        state = 'gameover';
        if (score > (window.__game?.highScore || 0)) {
          // optional highScore
        }
        beep(110, 0.25);
      } else {
        // safe passage
        score += 1;
        beep(880, 0.05);
      }
      // remove hazard
      hazards.splice(i, 1);
    }
  }
}
