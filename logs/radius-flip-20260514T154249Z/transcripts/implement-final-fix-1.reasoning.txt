function update(dt) {
  if (phase !== 'playing') return;

  // Update player size timers
  if (player.sizeTimer > 0) {
    player.sizeTimer -= dt;
    if (player.sizeTimer <= 0) {
      // Revert to small size and start cooldown
      player.size = 'small';
      player.radius = PLAYER_SMALL_R;
      player.cooldownTimer = SIZE_COOLDOWN;
      beep(660, 0.1);
    }
  }
  if (player.cooldownTimer > 0) {
    player.cooldownTimer -= dt;
  }

  // Auto-advance forward (simulated by moving hazards leftward)
  // Spawn hazards
  spawnTimer -= dt;
  if (spawnTimer <= 0) {
    spawnHazard();
    spawnTimer = HAZARD_SPAWN_MIN + Math.random() * (HAZARD_SPAWN_MAX - HAZARD_SPAWN_MIN);
  }

  // Update hazards
  for (let i = hazards.length - 1; i >= 0; i--) {
    const h = hazards[i];
    h.x -= HAZARD_SPEED * dt;
    // Collision with player
    const closestX = Math.max(h.x, Math.min(player.x, h.x + h.w));
    const closestY = Math.max(h.y, Math.min(player.y, h.y + h.h));
    const dx = player.x - closestX;
    const dy = player.y - closestY;
    const distSq = dx * dx + dy * dy;
    if (distSq < player.radius * player.radius) {
      if (player.size === 'large') {
        // Destroy hazard
        score += 10;
        beep(880, 0.05);
        hazards.splice(i, 1);
        continue;
      } else {
        // Player dies
        phase = 'gameover';
        beep(110, 0.5);
        // No need to process further hazards
        break;
      }
    }
    // Remove off-screen hazards
    if (h.x + h.w < 0) {
      hazards.splice(i, 1);
    }
  }

  // Increment score over time as a fallback
  score += Math.round(dt * 5);
}
