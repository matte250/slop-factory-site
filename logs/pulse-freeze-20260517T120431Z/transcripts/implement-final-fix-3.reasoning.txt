// === RESET ==============================================================
function reset() {
  player = { x: canvas.width / 2, y: canvas.height / 2, vx: 0, vy: 0, radius: 8 };
  hazards = [];
  score = 0;
  phase = "playing";
  keys.clear();
  lastTime = performance.now();
  fieldCooldown = 0;
  fieldTimer = 0;
  // spawn initial hazards
  for (let i = 0; i < 5; i++) spawnHazard();
  startLoop();
}

// Helper to spawn hazard away from player safe radius
function spawnHazard() {
  const SAFE_RADIUS = 120;
  let x, y;
  do {
    x = Math.random() * canvas.width;
    y = Math.random() * canvas.height;
  } while (Math.hypot(x - player.x, y - player.y) < SAFE_RADIUS);
  const angle = Math.random() * Math.PI * 2;
  const speed = HAZARD_SPEED;
  hazards.push({
    x,
    y,
    vx: Math.cos(angle) * speed,
    vy: Math.sin(angle) * speed,
    size: HAZARD_SIZE,
  });
}

let fieldTimer = 0;
let fieldCooldown = 0;
function update(dt) {
  if (phase !== "playing") return;
  // player input
  let ax = 0, ay = 0;
  if (keys.has("ArrowLeft") || keys.has("a") || keys.has("A")) ax -= 1;
  if (keys.has("ArrowRight") || keys.has("d") || keys.has("D")) ax += 1;
  if (keys.has("ArrowUp") || keys.has("w") || keys.has("W")) ay -= 1;
  if (keys.has("ArrowDown") || keys.has("s") || keys.has("S")) ay += 1;
  if (ax !== 0 || ay !== 0) {
    const len = Math.hypot(ax, ay);
    ax = (ax / len) * PLAYER_ACCEL;
    ay = (ay / len) * PLAYER_ACCEL;
  }
  player.vx += ax * dt;
  player.vy += ay * dt;
  // friction
  player.vx *= Math.pow(PLAYER_FRICTION, dt);
  player.vy *= Math.pow(PLAYER_FRICTION, dt);
  // clamp speed
  const sp = Math.hypot(player.vx, player.vy);
  if (sp > PLAYER_MAX_SPEED) {
    const s = PLAYER_MAX_SPEED / sp;
    player.vx *= s;
    player.vy *= s;
  }
  // move
  player.x = (player.x + player.vx * dt + canvas.width) % canvas.width;
  player.y = (player.y + player.vy * dt + canvas.height) % canvas.height;

  // field timers

function beep(freq, durSec) {
