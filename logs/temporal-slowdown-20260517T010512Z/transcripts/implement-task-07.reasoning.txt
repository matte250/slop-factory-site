function update(dt) {
  // Player movement
  dot.x += DOT_SPEED * dt;
  if (dot.x - DOT_RADIUS > canvas.width) dot.x = -DOT_RADIUS;

  // Hazard movement
  for (let i = hazards.length - 1; i >= 0; i--) {
    const hz = hazards[i];
    hz.x -= HAZARD_SPEED * dt;
    if (hz.x + HAZARD_RADIUS < 0) {
      hazards.splice(i, 1);
    }
  }

  // Spawn timer
  spawnTimer -= dt;
  if (spawnTimer <= 0) {
    spawnHazard();
    // reset timer to random interval
    spawnTimer = Math.random() * (MAX_SPAWN_INTERVAL - MIN_SPAWN_INTERVAL) + MIN_SPAWN_INTERVAL;
  }
}
