      // Reset game state
      player.x = canvas.width / 2;

          lastTime = now;

    function update(dt) {
      if (phase !== 'playing') return;

      // Apply anchor force
      const dir = anchorSide === 'left' ? -1 : 1;
      playerVx += dir * ANCHOR_FORCE * dt;
      // Clamp velocity
      if (playerVx > MAX_VELOCITY) playerVx = MAX_VELOCITY;
      if (playerVx < -MAX_VELOCITY) playerVx = -MAX_VELOCITY;
