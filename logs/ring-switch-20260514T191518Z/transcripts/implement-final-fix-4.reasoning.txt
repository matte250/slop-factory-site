function update(dt) {
        if (phase !== 'playing') return;
        // Store previous angle for scoring/gap detection
        const prevAngle = player.angle;
        // Auto-advance player along the tunnel
        player.angle += PLAYER_SPEED * dt;
        // Keep angle within 0 to 2π
        player.angle = (player.angle + Math.PI * 2) % (Math.PI * 2);
        // Collision detection with hazards
        for (const hazard of hazards) {
          if (hazard.ring !== player.ring) continue;
          const diff = Math.abs(((hazard.angle - player.angle + Math.PI) % (2 * Math.PI)) - Math.PI);
          if (diff < COLLISION_TOL) {
            // Game over! Stop the loop.
            phase = 'gameover';
            if (loopId !== null) cancelAnimationFrame(loopId);
            loopId = null;
            beep(110, 0.25);
            break;
          }
        }
        // Score increases over time
        score += dt * 10;
        // Spawn hazards periodically
        spawnTimer -= dt;
        if (spawnTimer <= 0) {
          const ring = Math.random() < 0.5 ? 0 : 1;
          const angle = player.angle + Math.PI * 2 + Math.random() * Math.PI;
          hazards.push({ ring, angle });
          spawnTimer = HAZARD_SPAWN_INTERVAL;
        }
        // Remove hazards that have passed behind the player
        hazards = hazards.filter(h => (h.angle > player.angle - Math.PI * 2));
      }

// === AUDIO (lazy) =======================================================
function beep(freq, durSec) {
  try {
    if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = 'square';
    osc.frequency.value = freq;
    gain.gain.setValueAtTime(0.05, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + durSec);
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + durSec);
  } catch (e) { /* ignore audio errors */ }
}
