function update(dt) {
    if (phase !== "playing") return;
    
    // Player forward motion (auto forward)
    player.y -= FORWARD_SPEED * dt;
    // optional lateral drift from keys
    const dirX = (keys.has("ArrowRight") || keys.has("d") || keys.has("D") ? 1 : 0) -
                 (keys.has("ArrowLeft") || keys.has("a") || keys.has("A") ? 1 : 0);
    player.x += dirX * LATERAL_SPEED * dt;

    // Keep player within canvas bounds
    player.x = clamp(player.x, PLAYER_LARGE_RADIUS, canvas.width - PLAYER_LARGE_RADIUS);
    player.y = clamp(player.y, PLAYER_SMALL_RADIUS, canvas.height - PLAYER_SMALL_RADIUS);
    
    // Spawn hazards
    spawnTimer -= dt;
    if (spawnTimer <= 0) {
        spawnTimer = SPAWN_INTERVAL;
        const hazard = {
            x: Math.random() * (canvas.width - 20) + 10,
            y: canvas.height + 20,
            radius: 12,
            // Could have type but not needed
        };
        // Ensure not within SAFE_RADIUS of player
        const dx = hazard.x - player.x;
        const dy = hazard.y - player.y;
        if (Math.hypot(dx, dy) > SAFE_RADIUS) {
            hazards.push(hazard);
        }
    }

    // Move hazards toward player (upward)
    for (let i = hazards.length - 1; i >= 0; i--) {
        const h = hazards[i];
        h.y -= FORWARD_SPEED * dt;
        // Check collision with player
