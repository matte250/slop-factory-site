function startLoop() {
  if (loopId !== null) cancelAnimationFrame(loopId);
  function tick(now) {
    try {
      const dt = Math.min((now - lastTime) / 1000, 0.1);
      lastTime = now;
      update(dt);
      // No render yet.
      publishState();
    } catch (err) {
      console.error("loop error", err);
    }
    loopId = requestAnimationFrame(tick);
  }
  loopId = requestAnimationFrame(tick);
}

function update(dt) {
  // bubble cooldown
  if (bubble.cooldown > 0) {
    bubble.cooldown -= dt;
    if (bubble.cooldown <= 0) {
      bubble.active = false;
    }
  }

  // update hazards
  for (const h of hazards) {
    const speed = h.speed * (bubble.active ? 0.3 : 1);
    h.x += speed * dt; // move horizontally (assuming left to right)
  }

  // collision detection

      phase = "gameover";
      beep(110, 0.25);
      if (loopId !== null) cancelAnimationFrame(loopId);
      break;
    }
  }
}
