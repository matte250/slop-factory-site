    // === AUDIO ==============================================================
    function beep(freq, durSec) {
      try {
        if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
        const osc = audioCtx.createOscillator();
        const gain = audioCtx.createGain();
        osc.type = "square";
        osc.frequency.value = freq;
        gain.gain.setValueAtTime(0.05, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + durSec);
        osc.connect(gain);
        gain.connect(audioCtx.destination);
        osc.start();
        osc.stop(audioCtx.currentTime + durSec);
      } catch {}
    }

    function update(dt) {
      if (phase !== "playing") return;
      // Player movement
      const dir = (keys.has("ArrowRight") || keys.has("d") || keys.has("D") ? 1 : 0) -
                  (keys.has("ArrowLeft")  || keys.has("a") || keys.has("A") ? 1 : 0);
      player.x += dir * PLAYER_SPEED * dt;
      // clamp
      player.x = Math.max(PLAYER_R, Math.min(canvas.width - PLAYER_R, player.x));

      // Bubble timer
      if (bubble.active) {
        bubble.timer -= dt;
        if (bubble.timer <= 0) {
          bubble.active = false;
          bubble.timer = 0;
        }
      }
      if (bubble.cooldown > 0) {
        bubble.cooldown -= dt;
        if (bubble.cooldown < 0) bubble.cooldown = 0;
      }

      // Spawn hazards
      spawnTimer -= dt;
      if (spawnTimer <= 0) {
        spawnHazard();
        spawnTimer = SPAWN_INTERVAL;
      }

      // Move hazards
      const speed = bubble.active ? WORLD_BASE_SPEED * BUBBLE_FACTOR : WORLD_BASE_SPEED;
      for (const hz of hazards) {
        hz.y += speed * dt;
      }

      // Remove off-screen hazards and scoring
      for (let i = hazards.length - 1; i >= 0; i--) {
        const hz = hazards[i];
        // Collision detection
        if (hz.x < player.x + PLAYER_R && hz.x + hz.w > player.x - PLAYER_R &&
            hz.y < player.y + PLAYER_R && hz.y + hz.h > player.y - PLAYER_R) {
          phase = "gameover";
          beep(200, 0.3);
          // Stop loop after this frame; not cancel now to let render overlay
        }

        // Scoring when hazard passes player
        if (!hz.passed && hz.y > player.y) {
          hz.passed = true;
          // Score: 1 point, plus bonus if bubble not active during pass
          score += bubble.active ? 1 : 2; // give bonus when not slowed
          beep(600, 0.05);
        }

        // Remove if off canvas
        if (hz.y > canvas.height) {
          hazards.splice(i, 1);
        }
      }
    }

    // === AUDIO ==============================================================
    function beep(freq, durSec) {
      try {
        if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
        const osc = audioCtx.createOscillator();
        const gain = audioCtx.createGain();
        osc.type = "square";
        osc.frequency.value = freq;
        gain.gain.setValueAtTime(0.05, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + durSec);
        osc.connect(gain);
        gain.connect(audioCtx.destination);
        osc.start();
        osc.stop(audioCtx.currentTime + durSec);
      } catch {}
    }

      if (loopId !== null) cancelAnimationFrame(loopId);

          lastTime = now;

    function update(dt) {
      if (phase !== "playing") return;

      // Player movement
      const dir = (keys.has("ArrowRight") || keys.has("d") || keys.has("D") ? 1 : 0) -
                  (keys.has("ArrowLeft") || keys.has("a") || keys.has("A") ? 1 : 0);
      player.x += dir * PLAYER_SPEED * dt;
      player.x = Math.max(PLAYER_R, Math.min(canvas.width - PLAYER_R, player.x));

      // Bubble timers
      if (bubble.active) {
        bubble.timer -= dt;
        if (bubble.timer <= 0) {
          bubble.active = false;
          bubble.timer = 0;
        }
      }
      if (bubble.cooldown > 0) {
        bubble.cooldown -= dt;
        if (bubble.cooldown < 0) bubble.cooldown = 0;
      }

      // Spawn hazards
      spawnTimer -= dt;
      if (spawnTimer <= 0) {
        spawnHazard();
        spawnTimer = SPAWN_INTERVAL;
      }

      // Move hazards
      const speed = bubble.active ? WORLD_BASE_SPEED * BUBBLE_FACTOR : WORLD_BASE_SPEED;
      for (const hz of hazards) {
        hz.y += speed * dt;
      }

      // Collision and scoring
      for (let i = hazards.length - 1; i >= 0; i--) {
        const hz = hazards[i];
        // Collision detection (point vs rect)
        if (
          player.x + PLAYER_R > hz.x &&
          player.x - PLAYER_R < hz.x + hz.w &&
          player.y + PLAYER_R > hz.y &&
          player.y - PLAYER_R < hz.y + hz.h

          beep(600, 0.05);

        }
      }
