    function update(dt) {
      // Player acceleration and forward motion
      player.speed = Math.min(PLAYER_MAX_SPEED, player.speed + PLAYER_ACCEL * dt);
      player.worldX += player.speed * dt;

      // Portal cooldown
      if (portalCooldown > 0) portalCooldown -= dt;

      // Spawn hazards
      spawnTimer -= dt;
      if (spawnTimer <= 0) {
        spawnHazard();
        spawnTimer = HAZARD_SPAWN_INTERVAL;
      }

      // Update hazards: check for collision and passing
      for (let i = hazards.length - 1; i >= 0; i--) {
        const h = hazards[i];
        // Compute on-screen X position
        const screenX = h.x - player.worldX + PLAYER_SCREEN_X;
        // Off left side: avoided
        if (screenX + h.w < 0) {
          hazards.splice(i, 1);
          score += 1; // avoided hazard
          continue;
        }
        // Collision check
        if (phase === "playing" && screenX < PLAYER_SCREEN_X + PLAYER_RADIUS && screenX + h.w > PLAYER_SCREEN_X - PLAYER_RADIUS) {
          // Simple collision on x; y aligned

      }
