function update(dt) {
  // Player movement
  const ACCEL = 200; // px/sec^2
  const MAX_SPEED = 300; // px/sec
  const FRICTION = 0.95; // per second (applied by Math.pow)
  if (player) {
    // apply acceleration based on keys
    if (keys.has('ArrowUp')) {
      player.vy -= ACCEL * dt;
    }
    if (keys.has('ArrowDown')) {
      player.vy += ACCEL * dt;
    }
    if (keys.has('ArrowLeft')) {
      player.vx -= ACCEL * dt;
    }
    if (keys.has('ArrowRight')) {
      player.vx += ACCEL * dt;
    }
    // apply friction to velocity
    const decay = Math.pow(FRICTION, dt);
    player.vx *= decay;
    player.vy *= decay;
    // clamp speed
    const speed = Math.hypot(player.vx, player.vy);
    if (speed > MAX_SPEED) {
      const scale = MAX_SPEED / speed;
      player.vx *= scale;
      player.vy *= scale;
    }
    // update position
    player.x = Math.max(0, Math.min(W, player.x + player.vx * dt));
    player.y = Math.max(0, Math.min(H, player.y + player.vy * dt));
  }
  // Update hazards movement
  for (const h of hazards) {
    h.x += h.vx * dt;
    h.y += h.vy * dt;
    // maybe wrap to opposite side? But spec says clamp, but hazards may also be removed if out of bounds?
    // For now, just keep them within bounds (wrap)
    if (h.x < 0) h.x += W;
    if (h.x > W) h.x -= W;
    if (h.y < 0) h.y += H;
    if (h.y > H) h.y -= H;
  }
}
