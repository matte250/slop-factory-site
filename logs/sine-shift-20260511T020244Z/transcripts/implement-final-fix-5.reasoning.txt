function startLoop() {
  if (loopId !== null) cancelAnimationFrame(loopId);
  function tick(now) {
    try {
      const dt = Math.min((now - lastTime) / 1000, 0.1);
      lastTime = now;
      update(dt);
      render();
      publishState();
    } catch (err) {
      console.error('loop error', err);
    }
    loopId = requestAnimationFrame(tick);
  }
  loopId = requestAnimationFrame(tick);
}

function update(dt) {
  if (state !== 'playing') return;
  // Advance player horizontally
  player.x += PLAYER_SPEED_X * dt;
  // Keep moving in periodic sine wave: y = middle + amplitude * sin(phase)
  player.phase += dt * 2 * Math.PI * 0.4; // 0.4 Hz maybe.
  player.y = CANVAS_H/2 + WAVE_AMPLITUDE * Math.sin(player.phase);
  // If player passes canvas width, wrap or stop? For simplicity, wrap back to left.
  if (player.x > CANVAS_W + PLAYER_RADIUS) {
    player.x = -PLAYER_RADIUS;
  }
  // Input: shift phase instantly on Space press
  if (keys.has(' ') || keys.has('Space')) {
    // Shift phase by PI/2 upward or downward? We'll toggle up/down.
    // If currently sin positive, shift to negative? But simple: add Phase shift.

  }
}

function beep(freq, durSec) {
  try {
    if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = 'square';
    osc.frequency.value = freq;
    gain.gain.setValueAtTime(0.05, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + durSec);
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + durSec);
  } catch (e) {
    // ignore
  }
}
