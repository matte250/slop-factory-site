```js
// === CONFIG =============================================================
const CANVAS_W = 800;

const WAVE_AMPLITUDE = 80;
const WAVE_FREQUENCY = 2 * Math.PI / 200; // per pixel? Actually rad per pixel.
```

State:

```js
// === STATE ==============================================================
let canvas, ctx;
let player; // { x, y, phase }

let state = 'menu'; // 'playing'|'gameover'
let keys = new Set();
let loopId = null;
```

Reset:

```js
function reset() {
  player = { x: 0, y: CANVAS_H/2, phase: 0 };
  score = 0;
  state = 'playing';
  keys.clear();
  startLoop();
}
```

Loop:

```js
function startLoop() {
  if (loopId !== null) cancelAnimationFrame(loopId);
  let lastTime = performance.now();
  function tick(now) {
    try {
      const dt = Math.min((now - lastTime) / 1000, 0.1);
      lastTime = now;
      update(dt);
      render();
      publishState();
    } catch (e) {
      console.error('loop error', e);
    }
    loopId = requestAnimationFrame(tick);
  }
  loopId = requestAnimationFrame(tick);
}
```

Update:

```js
function update(dt) {
  if (state !== 'playing') return;
  // Move player horizontally
  player.x += PLAYER_SPEED * dt;
  // Wrap around or stop at canvas width? Let's keep moving.
  // Compute vertical position based on sine wave
  player.y = CANVAS_H/2 + Math.sin(player.x * WAVE_FREQUENCY + player.phase) * WAVE_AMPLITUDE;
  // Example scoring: increase over time
  score = Math.floor(player.x / 10);
}
```

Render:

```js
function render() {
  ctx.fillStyle = '#0f1320';
  ctx.fillRect(0, 0, CANVAS_W, CANVAS_H);
  // Draw player dot
  ctx.fillStyle = '#ff00aa';
  ctx.beginPath();
  ctx.arc(player.x, player.y, PLAYER_R, 0, Math.PI*2);
  ctx.fill();
  // HUD
  ctx.fillStyle = '#e6e6f0';
  ctx.font = '16px ui-sans-serif, system-ui';
  ctx.textAlign = 'left';
  ctx.fillText('Score ' + score, 12, 22);
}
```
