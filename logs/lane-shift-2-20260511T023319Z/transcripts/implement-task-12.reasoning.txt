```
function update(dt) {
  if (phase !== 'playing') return;
  // update safe timer and spawn timer
  safeTimer = Math.max(0, safeTimer - dt);
  hazardSpawnTimer += dt;
  if (safeTimer <= 0 && hazardSpawnTimer >= HAZARD_SPAWN_INTERVAL) {
    spawnHazard();
    hazardSpawnTimer = 0;
  }

  // move hazards
  for (let i = hazards.length - 1; i >= 0; i--) {
    const h = hazards[i];
    h.y += HAZARD_SPEED * dt;
    // check collision
    if (h.lane === player.lane && h.y + HAZARD_RADIUS >= player.y - PLAYER_RADIUS && h.y - HAZARD_RADIUS <= player.y + PLAYER_RADIUS) {
      // collision
      phase = 'gameover';
      beep(120, 0.18);
      // stop loop? we will stop spawning etc.
      // keep hazard for rendering maybe
      break;
    }
    // if passed player (i.e., beyond bottom)
    if (h.y - HAZARD_RADIUS > canvas.height) {
      // survived
      hazards.splice(i, 1);
      score++;
      beep(660, 0.06);
    }
  }
}
```

```
function beep(freq, durSec) {
  try {
    if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = 'square';
    osc.frequency.value = freq;
    gain.gain.setValueAtTime(0.06, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + durSec);
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + durSec);
  } catch {}
}
```

```
  // === AUDIO (lazy) =======================================================
  function beep(freq, durSec) {
    try {
      if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
      const osc = audioCtx.createOscillator();
      const gain = audioCtx.createGain();
      osc.type = 'square';
      osc.frequency.value = freq;
      gain.gain.setValueAtTime(0.06, audioCtx.currentTime);
      gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + durSec);
      osc.connect(gain);
      gain.connect(audioCtx.destination);
      osc.start();
      osc.stop(audioCtx.currentTime + durSec);
    } catch {}
  }
```
