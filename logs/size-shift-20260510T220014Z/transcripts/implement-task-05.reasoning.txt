    try {
      const dt = Math.min((now - lastTime) / 1000, 0.1);
      lastTime = now;
      // Update player position based on keys.
      // We'll incorporate movement update directly here or call update().
      const speed = 300; // px/sec as placeholder.
      let dirX = 0, dirY = 0;
      if (keys.has('ArrowRight') || keys.has('d') || keys.has('D')) dirX += 1;
      if (keys.has('ArrowLeft') || keys.has('a') || keys.has('A')) dirX -= 1;
      if (keys.has('ArrowDown') || keys.has('s') || keys.has('S')) dirY += 1;
      if (keys.has('ArrowUp') || keys.has('w') || keys.has('W')) dirY -= 1;
      const len = Math.hypot(dirX, dirY);
      let normX = 0, normY = 0;
      if (len > 0) { normX = dirX / len; normY = dirY / len; }
      player.x += normX * speed * dt;
      player.y += normY * speed * dt;
      // Clamp
      player.x = Math.max(player.radius, Math.min(canvas.width - player.radius, player.x));
      player.y = Math.max(player.radius, Math.min(canvas.height - player.radius, player.y));

      ctx.beginPath();

      ctx.fill();
