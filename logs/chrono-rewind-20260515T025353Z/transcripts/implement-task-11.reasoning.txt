  function startLoop() {
    if (loopId !== null) cancelAnimationFrame(loopId);
    function tick(now) {
      try {
        const dt = Math.min((now - lastTime) / 1000, 0.1);
        lastTime = now;
        update(dt);
        render();
        publishState();
      } catch (err) {

      }
      loopId = requestAnimationFrame(tick);
    }
    loopId = requestAnimationFrame(tick);

  function update(dt) {
    if (phase !== "playing") return;

    // Player control
    let accel = 0;
    if (keys.has("ArrowRight") || keys.has("d") || keys.has("D")) accel += PLAYER_ACCEL;
    if (keys.has("ArrowLeft") || keys.has("a") || keys.has("A")) accel -= PLAYER_ACCEL;
    player.vx += accel * dt;
    // Apply friction
    const decay = Math.pow(PLAYER_FRICTION, dt);
    player.vx *= decay;
    // Clamp speed
    if (player.vx > PLAYER_MAX_SPEED) player.vx = PLAYER_MAX_SPEED;
    if (player.vx < -PLAYER_MAX_SPEED) player.vx = -PLAYER_MAX_SPEED;

    // Rewind handling
    if (rewindTimer > 0) {
      // move entities backwards
      player.x = wrap(player.x - player.vx * dt, W);
      for (const h of hazards) {
        h.x = wrap(h.x - h.vx * dt, W);
        h.y = wrap(h.y - h.vy * dt, H);
      }
      rewindTimer = Math.max(0, rewindTimer - dt);
      if (rewindTimer === 0) {
        // start cooldown
        rewindCooldown = REWIND_COOLDOWN;
        beep(440, 0.1);
      }
    } else {
      // Normal forward update
      player.x = wrap(player.x + player.vx * dt, W);
      // Spawn hazards
      spawnTimer -= dt;
      if (spawnTimer <= 0) {
        spawnHazard();
        spawnTimer = HAZARD_SPAWN_INTERVAL;
      }
      // Update hazards
      for (const h of hazards) {
        h.x = wrap(h.x + h.vx * dt, W);
        h.y += h.vy * dt;
      }
      // Remove off‑screen hazards and award score
      hazards = hazards.filter((h) => {
        if (h.y - h.radius > H) {
          score += SCORE_PER_HAZARD;
          return false;
        }
        return true;
      });
      // Decrease cooldown timer if active
      if (rewindCooldown > 0) {
        rewindCooldown = Math.max(0, rewindCooldown - dt);
      }
    }

    // Collision detection
    for (const h of hazards) {
      const dist = Math.hypot(player.x - h.x, player.y - h.y);
      if (dist < player.radius + h.radius) {
        phase = "gameover";
        if (score > highScore) highScore = score;
        beep(110, 0.3);
        break;
      }
    }
  }

      tries++;
    } while (Math.abs(x - player.x) < PLAYER_RADIUS * 4 && tries < 30);
    const speed = HAZARD_SPEED_RANGE[0] + Math.random() * (HAZARD_SPEED_RANGE[1] - HAZARD_SPEED_RANGE[0]);
    hazards.push({
