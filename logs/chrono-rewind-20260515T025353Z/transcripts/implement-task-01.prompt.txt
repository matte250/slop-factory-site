## Reference example: "asteroids"

The following is a complete, validator-passing game. **Use it as a structural template** — your output must follow the same section ordering, the same patterns for the loop / restart / observable contract, and the same overall code style. Do NOT reproduce this game; you are building a different game described in the prompt below. The example exists only to anchor structure.

Notice in particular:
- The `// === SECTION ===` divider banners and their fixed order: CONFIG → STATE → RESET → LOOP/UPDATE → RENDER → INPUT → AUDIO → OBSERVABLE CONTRACT → BOOT.
- Plain object literals for game state (no `class`).
- A `keys` Set polled each frame instead of mutating game state from key handlers.
- `if (loopId !== null) cancelAnimationFrame(loopId)` at the top of `startLoop()`.
- The animation loop body wrapped in `try/catch`.
- `publishState()` called once per frame at the end of the tick.
- `window.__game` populated honestly (player.visible reflects actual viewport position).
- AudioContext initialized lazily inside `beep()`, wrapped in try/catch.

### Example `index.html`
```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="utf-8">
  <title>Neon Asteroids</title>
  <style>
    body { margin: 0; background: #0a0a0f; color: #e6e6f0; font-family: ui-sans-serif, system-ui, sans-serif; }
    .game-header { padding: 12px 16px; border-bottom: 1px solid #1d1d2a; }
    .game-header h1 { margin: 0; font-size: 18px; color: #7df9ff; }
    .game-header .game-description { margin: 4px 0; color: #a5a5b8; }
    .game-header .game-controls { margin: 4px 0; padding-left: 18px; color: #c8c8d8; }
    main { display: flex; flex-direction: column; align-items: center; padding: 16px; }
    canvas { background: #05050c; border: 1px solid #2a2f44; }
    .game-restart { margin-top: 10px; padding: 8px 14px; background: #1c2030; color: #7df9ff; border: 1px solid #2a2f44; cursor: pointer; }
    .game-restart:hover { background: #232842; }
  </style>
</head>
<body>
  <header class="game-header">
    <h1>Neon Asteroids</h1>
    <p class="game-description">Rotate, thrust, and shoot. Big rocks split into small ones. Get hit and you're gone.</p>
    <ul class="game-controls">
      <li>Left/Right (or A/D) to rotate</li>
      <li>Up (or W) to thrust</li>
      <li>Space to fire</li>
      <li>Press R or click Restart</li>
    </ul>
  </header>
  <main>
    <canvas id="game" width="800" height="600"></canvas>
    <button id="restart" class="game-restart" type="button">Restart</button>
  </main>
  <script>
  // === CONFIG =============================================================
  const W = 800, H = 600;
  const SHIP_RADIUS = 12;
  const SHIP_THRUST = 280;       // px/sec^2
  const SHIP_FRICTION = 0.6;     // velocity decay per second (1 = none)
  const SHIP_TURN = Math.PI * 1.8; // rad/sec
  const BULLET_SPEED = 520;      // px/sec
  const BULLET_LIFE = 1.0;       // sec
  const FIRE_COOLDOWN = 0.20;    // sec between shots
  const ASTEROID_BIG_R = 38;
  const ASTEROID_MID_R = 22;
  const ASTEROID_LIL_R = 12;
  const ASTEROID_SPEED_RANGE = [40, 110];
  const SAFE_RADIUS_AT_SPAWN = 160; // no asteroid spawns inside this disk

  const SCORE_BIG = 20, SCORE_MID = 50, SCORE_LIL = 100;

  // === STATE ==============================================================
  const canvas = document.getElementById("game");
  const ctx = canvas.getContext("2d");
  const restartBtn = document.getElementById("restart");

  let ship;        // { x, y, vx, vy, angle, alive }
  let bullets;     // [{ x, y, vx, vy, life }]
  let asteroids;   // [{ x, y, vx, vy, r, shape: number[] }]
  let score;
  let highScore = 0;
  let phase;       // "playing" | "gameover"
  let fireCooldown;
  let keys;
  let lastTime;
  let loopId = null;
  let audioCtx = null;

  function reset() {
    ship = { x: W / 2, y: H / 2, vx: 0, vy: 0, angle: -Math.PI / 2, alive: true };
    bullets = [];
    asteroids = [];
    spawnWave(5);
    score = 0;
    phase = "playing";
    fireCooldown = 0;
    keys = new Set();
    lastTime = performance.now();
    publishState();
    startLoop();
  }

  function startLoop() {
    if (loopId !== null) cancelAnimationFrame(loopId);
    function tick(now) {
      try {
        const dt = Math.min((now - lastTime) / 1000, 0.1);
        lastTime = now;
        update(dt);
        render();
        publishState();
      } catch (err) {
        console.error("loop error", err);
      }
      loopId = requestAnimationFrame(tick);
    }
    loopId = requestAnimationFrame(tick);
  }

  function spawnWave(n) {
    for (let i = 0; i < n; i++) spawnAsteroid(ASTEROID_BIG_R);
  }

  function spawnAsteroid(r) {
    let x, y, tries = 0;
    do {
      x = Math.random() * W;
      y = Math.random() * H;
      tries++;
    } while (Math.hypot(x - ship.x, y - ship.y) < SAFE_RADIUS_AT_SPAWN && tries < 30);
    const angle = Math.random() * Math.PI * 2;
    const speed = ASTEROID_SPEED_RANGE[0] + Math.random() * (ASTEROID_SPEED_RANGE[1] - ASTEROID_SPEED_RANGE[0]);
    asteroids.push({
      x, y,
      vx: Math.cos(angle) * speed,
      vy: Math.sin(angle) * speed,
      r,
      shape: makeAsteroidShape(),
    });
  }

  function makeAsteroidShape() {
    // Per-vertex radius multipliers, 8-12 vertices for a knobbly silhouette.
    const n = 8 + Math.floor(Math.random() * 5);
    const out = [];
    for (let i = 0; i < n; i++) out.push(0.75 + Math.random() * 0.45);
    return out;
  }

  // === UPDATE =============================================================
  function update(dt) {
    if (phase !== "playing") return;

    // Ship.
    if (ship.alive) {
      const turnIn = (keys.has("ArrowRight") || keys.has("d") || keys.has("D") ? 1 : 0)
                   - (keys.has("ArrowLeft")  || keys.has("a") || keys.has("A") ? 1 : 0);
      ship.angle += turnIn * SHIP_TURN * dt;

      const thrusting = keys.has("ArrowUp") || keys.has("w") || keys.has("W");
      if (thrusting) {
        ship.vx += Math.cos(ship.angle) * SHIP_THRUST * dt;
        ship.vy += Math.sin(ship.angle) * SHIP_THRUST * dt;
      }
      // Mild friction so the ship doesn't drift forever.
      const decay = Math.pow(SHIP_FRICTION, dt);
      ship.vx *= decay;
      ship.vy *= decay;
      ship.x = wrap(ship.x + ship.vx * dt, W);
      ship.y = wrap(ship.y + ship.vy * dt, H);

      fireCooldown -= dt;
      if ((keys.has(" ") || keys.has("Space")) && fireCooldown <= 0) {
        fire();
        fireCooldown = FIRE_COOLDOWN;
      }
    }

    // Bullets.
    for (const b of bullets) {
      b.life -= dt;
      b.x = wrap(b.x + b.vx * dt, W);
      b.y = wrap(b.y + b.vy * dt, H);
    }
    bullets = bullets.filter((b) => b.life > 0);

    // Asteroids.
    for (const a of asteroids) {
      a.x = wrap(a.x + a.vx * dt, W);
      a.y = wrap(a.y + a.vy * dt, H);
    }

    // Bullet-asteroid collisions.
    for (let i = asteroids.length - 1; i >= 0; i--) {
      const a = asteroids[i];
      let hit = false;
      for (let j = bullets.length - 1; j >= 0; j--) {
        const b = bullets[j];
        if (Math.hypot(a.x - b.x, a.y - b.y) < a.r) {
          bullets.splice(j, 1);
          hit = true;
          break;
        }
      }
      if (hit) splitAsteroid(i);
    }

    // Ship-asteroid collisions.
    if (ship.alive) {
      for (const a of asteroids) {
        if (Math.hypot(a.x - ship.x, a.y - ship.y) < a.r + SHIP_RADIUS * 0.6) {
          ship.alive = false;
          phase = "gameover";
          if (score > highScore) highScore = score;
          beep(110, 0.3);
          break;
        }
      }
    }

    // Wave clear -> spawn next.
    if (asteroids.length === 0 && phase === "playing") {
      spawnWave(5);
    }
  }

  function fire() {
    bullets.push({
      x: ship.x + Math.cos(ship.angle) * SHIP_RADIUS,
      y: ship.y + Math.sin(ship.angle) * SHIP_RADIUS,
      vx: ship.vx + Math.cos(ship.angle) * BULLET_SPEED,
      vy: ship.vy + Math.sin(ship.angle) * BULLET_SPEED,
      life: BULLET_LIFE,
    });
    beep(820, 0.04);
  }

  function splitAsteroid(idx) {
    const a = asteroids[idx];
    asteroids.splice(idx, 1);
    if (a.r === ASTEROID_BIG_R) {
      score += SCORE_BIG;
      for (let i = 0; i < 2; i++) addChild(a, ASTEROID_MID_R);
      beep(440, 0.06);
    } else if (a.r === ASTEROID_MID_R) {
      score += SCORE_MID;
      for (let i = 0; i < 2; i++) addChild(a, ASTEROID_LIL_R);
      beep(560, 0.05);
    } else {
      score += SCORE_LIL;
      beep(700, 0.05);
    }
  }

  function addChild(parent, r) {
    const angle = Math.random() * Math.PI * 2;
    const speed = ASTEROID_SPEED_RANGE[0] + Math.random() * (ASTEROID_SPEED_RANGE[1] - ASTEROID_SPEED_RANGE[0]);
    asteroids.push({
      x: parent.x, y: parent.y,
      vx: Math.cos(angle) * speed,
      vy: Math.sin(angle) * speed,
      r,
      shape: makeAsteroidShape(),
    });
  }

  // === RENDER =============================================================
  function render() {
    ctx.fillStyle = "#05050c";
    ctx.fillRect(0, 0, W, H);

    // Asteroids.
    ctx.strokeStyle = "#a5a5b8";
    ctx.lineWidth = 1.5;
    for (const a of asteroids) drawAsteroid(a);

    // Bullets.
    ctx.fillStyle = "#ffd23f";
    for (const b of bullets) {
      ctx.beginPath();
      ctx.arc(b.x, b.y, 2.4, 0, Math.PI * 2);
      ctx.fill();
    }

    // Ship.
    if (ship.alive) drawShip();

    // HUD.
    ctx.fillStyle = "#e6e6f0";
    ctx.font = "16px ui-sans-serif, system-ui";
    ctx.textAlign = "left";
    ctx.fillText(`Score ${score}`, 12, 22);
    ctx.textAlign = "right";
    ctx.fillText(`Best ${highScore}`, W - 12, 22);

    if (phase === "gameover") {
      ctx.fillStyle = "rgba(5,5,12,0.65)";
      ctx.fillRect(0, 0, W, H);
      ctx.textAlign = "center";
      ctx.font = "44px ui-sans-serif, system-ui";
      ctx.fillStyle = "#ff52ae";
      ctx.fillText("Game Over", W / 2, H / 2 - 20);
      ctx.fillStyle = "#e6e6f0";
      ctx.font = "20px ui-sans-serif, system-ui";
      ctx.fillText(`Final score ${score}`, W / 2, H / 2 + 14);
      ctx.fillStyle = "#a5a5b8";
      ctx.font = "16px ui-sans-serif, system-ui";
      ctx.fillText("Press R or click Restart", W / 2, H / 2 + 42);
    }
  }

  function drawShip() {
    ctx.save();
    ctx.translate(ship.x, ship.y);
    ctx.rotate(ship.angle);
    ctx.strokeStyle = "#7df9ff";
    ctx.lineWidth = 2;
    ctx.beginPath();
    ctx.moveTo(SHIP_RADIUS, 0);
    ctx.lineTo(-SHIP_RADIUS * 0.9, SHIP_RADIUS * 0.7);
    ctx.lineTo(-SHIP_RADIUS * 0.5, 0);
    ctx.lineTo(-SHIP_RADIUS * 0.9, -SHIP_RADIUS * 0.7);
    ctx.closePath();
    ctx.stroke();
    if (keys && (keys.has("ArrowUp") || keys.has("w") || keys.has("W"))) {
      ctx.strokeStyle = "#ffd23f";
      ctx.beginPath();
      ctx.moveTo(-SHIP_RADIUS * 0.5, 0);
      ctx.lineTo(-SHIP_RADIUS - 6 - Math.random() * 4, 0);
      ctx.stroke();
    }
    ctx.restore();
  }

  function drawAsteroid(a) {
    ctx.beginPath();
    for (let i = 0; i < a.shape.length; i++) {
      const t = (i / a.shape.length) * Math.PI * 2;
      const r = a.r * a.shape[i];
      const x = a.x + Math.cos(t) * r;
      const y = a.y + Math.sin(t) * r;
      if (i === 0) ctx.moveTo(x, y); else ctx.lineTo(x, y);
    }
    ctx.closePath();
    ctx.stroke();
  }

  // === INPUT ==============================================================
  document.addEventListener("keydown", (ev) => {
    if (ev.key === "r" || ev.key === "R") { reset(); return; }
    keys.add(ev.key);
    if (ev.key === " ") ev.preventDefault();
  });
  document.addEventListener("keyup", (ev) => { keys.delete(ev.key); });
  restartBtn.addEventListener("click", reset);

  // === AUDIO (lazy) =======================================================
  function beep(freq, durSec) {
    try {
      if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
      const osc = audioCtx.createOscillator();
      const gain = audioCtx.createGain();
      osc.type = "triangle";
      osc.frequency.value = freq;
      gain.gain.setValueAtTime(0.04, audioCtx.currentTime);
      gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + durSec);
      osc.connect(gain); gain.connect(audioCtx.destination);
      osc.start(); osc.stop(audioCtx.currentTime + durSec);
    } catch {}
  }

  // === OBSERVABLE CONTRACT ===============================================
  function publishState() {
    window.__game = {
      state: phase,                                  // "playing" | "gameover"
      score,
      player: {
        x: ship.x,
        y: ship.y,
        // wrap() keeps coords in [0, W) and [0, H), so visible is always true
        // — but assert it explicitly so the contract is honest.
        visible: ship.x >= 0 && ship.x < W && ship.y >= 0 && ship.y < H,
      },
      objective: "Shoot the asteroids before they hit your ship. Big rocks split into smaller ones.",
    };
  }

  // === HELPERS ============================================================
  function wrap(v, max) {
    if (v < 0) return v + max;
    if (v >= max) return v - max;
    return v;
  }

  // === BOOT ===============================================================
  reset();
  </script>
</body>
</html>
```
