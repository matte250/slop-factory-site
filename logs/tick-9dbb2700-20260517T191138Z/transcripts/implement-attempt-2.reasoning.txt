```js
const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');
canvas.width = 400; canvas.height = 600;

const ship = { x: canvas.width/2, y: canvas.height-30, w: 40, h: 20, speed:5 };
let asteroids = [];
let lastSpawn = 0;
let score = 0;
let lives = 3;
let keys = {};

function spawnAsteroid() {
  const size = 20 + Math.random()*30;
  asteroids.push({ x: Math.random()*(canvas.width-size), y:-size, w:size, h:size, speed: 2+Math.random()*3 });
}

function update(dt){
  // input
  if(keys['ArrowLeft']) ship.x -= ship.speed;
  if(keys['ArrowRight']) ship.x += ship.speed;
  ship.x = Math.max(0, Math.min(canvas.width-ship.w, ship.x));

  // asteroids
  for(let a of asteroids){
    a.y += a.speed;
  }
  // remove off-screen
  asteroids = asteroids.filter(a=>a.y < canvas.height);
  // spawn
  if(Date.now() - lastSpawn > 800){
    spawnAsteroid();
    lastSpawn = Date.now();
  }
  // collisions
  for(let i=asteroids.length-1;i>=0;i--){
    const a=asteroids[i];
    if(!(ship.x > a.x + a.w || ship.x + ship.w < a.x || ship.y > a.y + a.h || ship.y + ship.h < a.y)){
      lives--;
      asteroids.splice(i,1);
      if(lives===0) {cancelAnimationFrame(id); alert('Game Over');}
    }
  }
  score += dt/1000;
}

function draw(){
  ctx.clearRect(0,0,canvas.width,canvas.height);
  // ship
  ctx.fillStyle='white';
  ctx.fillRect(ship.x, ship.y, ship.w, ship.h);
  // asteroids
  ctx.fillStyle='gray';
  for(let a of asteroids){
    ctx.fillRect(a.x, a.y, a.w, a.h);
  }
  // UI
  ctx.fillStyle='yellow';
  ctx.font='16px sans-serif';
  ctx.fillText(`Score: ${Math.floor(score)}`,10,20);
  ctx.fillText(`Lives: ${lives}`,10,40);
}

let last = 0;
let id;
function loop(timestamp){
  const dt = timestamp - last;
  last = timestamp;
  update(dt);
  draw();
  if(lives>0) id = requestAnimationFrame(loop);
}
requestAnimationFrame(loop);

window.addEventListener('keydown',e=>keys[e.key]=true);
window.addEventListener('keyup',e=>keys[e.key]=false);
```
