```js
// === LOOP ==============================================================
function startLoop() {
  if (loopId !== null) cancelAnimationFrame(loopId);
  function tick(now) {
    try {
      const dt = Math.min((now - lastTime) / 1000, 0.1);
      lastTime = now;
      update(dt);
      render();
      publishState();
    } catch (err) {
      console.error('loop error', err);
    }
    loopId = requestAnimationFrame(tick);
  }
  loopId = requestAnimationFrame(tick);
}
```

Update:

```js
// === UPDATE =============================================================
function update(dt) {
  if (phase !== 'playing') return;

  // Player control
  let ax = 0, ay = 0;
  if (keys.has('ArrowLeft') || keys.has('a') || keys.has('A')) ax -= 1;
  if (keys.has('ArrowRight') || keys.has('d') || keys.has('D')) ax += 1;
  if (keys.has('ArrowUp') || keys.has('w') || keys.has('W')) ay -= 1;
  if (keys.has('ArrowDown') || keys.has('s') || keys.has('S')) ay += 1;
  // Normalize diagonal speed
  if (ax !== 0 && ay !== 0) {
    const inv = 1 / Math.sqrt(2);
    ax *= inv;
    ay *= inv;
  }
  player.vx = ax * PLAYER_SPEED;
  player.vy = ay * PLAYER_SPEED;

  // Apply friction (optional)
  player.vx *= Math.pow(PLAYER_FRICTION, dt);
  player.vy *= Math.pow(PLAYER_FRICTION, dt);

  // Update position
  player.x = Math.max(0, Math.min(W, player.x + player.vx * dt));
  player.y = Math.max(0, Math.min(H, player.y + player.vy * dt));

  // Slip cooldown timer
  slipCooldown = Math.max(0, slipCooldown - dt);

  // Hazards movement and removal
  for (const h of hazards) {
    h.x += h.vx * dt;
    h.y += h.vy * dt;
  }
  hazards = hazards.filter(h => h.x + HAZARD_R > 0 && h.y + HAZARD_R > 0 && h.x - HAZARD_R < W && h.y - HAZARD_R < H);

  // Spawn logic
  spawnTimer -= dt;
  if (spawnTimer <= 0) {
    spawnHazard();
    spawnTimer = SPAWN_INTERVAL;
  }

  // Collision detection (task 8)
  if (player.alive) {
    for (const h of hazards) {
      const dist = Math.hypot(player.x - h.x, player.y - h.y);
      if (dist < PLAYER_R + HAZARD_R) {
        player.alive = false;
        phase = 'gameover';
        if (score > highScore) highScore = score;
        beep(110, 0.3);
        break;
      }
    }
  }

  // Score maybe (increment over time)
  score = Math.floor(score + dt * 10); // just for test
}
```

```js
function spawnHazard() {
  // spawn at right edge
  let x = W + HAZARD_R;
  let y = HAZARD_R + Math.random() * (H - 2 * HAZARD_R);
  // ensure safe distance from player at spawn
  const maxTries = 30;
  let tries = 0;
  while (Math.hypot(x - player.x, y - player.y) < SAFE_RADIUS && tries < maxTries) {
    y = HAZARD_R + Math.random() * (H - 2 * HAZARD_R);
    tries++;
  }
  hazards.push({ x, y, vx: -HAZARD_SPEED, vy: 0, r: HAZARD_R });
}
```

```js
function render() {
  ctx.fillStyle = '#05050c';
  ctx.fillRect(0, 0, W, H);

  // Draw hazards as spikes (simple triangle)
  ctx.fillStyle = '#ff5252';
  for (const h of hazards) {
    ctx.beginPath();
    ctx.moveTo(h.x, h.y - h.r);
    ctx.lineTo(h.x - h.r, h.y + h.r);
```

```js
// === AUDIO (lazy) =======================================================
function beep(freq, durSec) {
  try {
    if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = 'triangle';
    osc.frequency.value = freq;
    gain.gain.setValueAtTime(0.04, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + durSec);
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + durSec);
  } catch (e) {
    // ignore audio errors
  }
}
```
