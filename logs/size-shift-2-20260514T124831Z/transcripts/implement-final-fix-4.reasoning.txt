function update(dt) {
  if (phase !== 'playing') return;

  // Player movement
  const moveLeft = keys.has('ArrowLeft') || keys.has('a') || keys.has('A');
  const moveRight = keys.has('ArrowRight') || keys.has('d') || keys.has('D');
  const dir = (moveRight ? 1 : 0) - (moveLeft ? 1 : 0);
  player.x += dir * PLAYER_SPEED * dt;
  // Clamp within canvas
  player.x = clamp(player.x, player.radius * player.sizeMult, canvas.width - player.radius * player.sizeMult);
  
  // Size shift timer
  if (sizeShiftTimer > 0) {
    sizeShiftTimer -= dt;
    if (sizeShiftTimer <= 0) {
      player.sizeMult = 1;
    }
  }

  // Hazard spawning
  hazardSpawnTimer -= dt;
  if (hazardSpawnTimer <= 0) {
    // spawn hazard off right edge
    const y = Math.random() * (canvas.height - 2 * HAZARD_RADIUS) + HAZARD_RADIUS;
    hazards.push({ x: canvas.width + HAZARD_RADIUS, y, radius: HAZARD_RADIUS });
    hazardSpawnTimer = SPAWN_INTERVAL;
  }

  // Move hazards
  for (let i = hazards.length - 1; i >= 0; i--) {
    const h = hazards[i];
    h.x -= HAZARD_SPEED * dt;
    // Collision detection
    const dx = h.x - player.x;
    const dy = h.y - player.y;
    const dist = Math.hypot(dx, dy);
    if (dist < (h.radius + player.radius * player.sizeMult)) {
      phase = 'gameover';
      beep(110, 0.25);
      // keep hazard for rendering
    } else if (h.x + h.radius < 0) {
      // Hazard passed, increase score
      score += 1;
      beep(880, 0.05);
      hazards.splice(i, 1);
    }
  }
}

// === RENDER =============================================================
function render() {
  // Clear background
  ctx.fillStyle = "#0f1320";
  ctx.fillRect(0, 0, canvas.width, canvas.height);
  
  // Draw hazards
  ctx.fillStyle = "#ff52ae";
  for (const h of hazards) {
    ctx.beginPath();
    ctx.arc(h.x, h.y, h.radius, 0, Math.PI * 2);
    ctx.fill();
  }
  
  // Draw player
  ctx.fillStyle = "#7df9ff";
  ctx.beginPath();
  ctx.arc(player.x, player.y, player.radius * player.sizeMult, 0, Math.PI * 2);
  ctx.fill();
  
  // HUD
  ctx.fillStyle = "#e6e6f0";
  ctx.font = "16px ui-sans-serif, system-ui";
  ctx.textAlign = "left";
  ctx.fillText(`Score ${score}`, 12, 22);
  
  // Game over overlay

function update(dt) {
  if (phase !== "playing") return;

  // size shift timer
  if (sizeShiftTimer > 0) {
    sizeShiftTimer -= dt;
    if (sizeShiftTimer <= 0) {
      player.sizeMult = 1;
      sizeShiftMode = "normal";
      sizeShiftTimer = 0;
    }
  }

  // movement
  let dirX = 0, dirY = 0;
  if (keys.has("ArrowLeft") || keys.has("a") || keys.has("A")) dirX -= 1;
  if (keys.has("ArrowRight") || keys.has("d") || keys.has("D")) dirX += 1;
  if (keys.has("ArrowUp") || keys.has("w") || keys.has("W")) dirY -= 1;
  if (keys.has("ArrowDown") || keys.has("s") || keys.has("S")) dirY += 1;
  if (dirX !== 0 || dirY !== 0) {
    const len = Math.hypot(dirX, dirY);
    dirX /= len;
    dirY /= len;
    player.x += dirX * PLAYER_SPEED * dt;
    player.y += dirY * PLAYER_SPEED * dt;
    // clamp
    player.x = Math.max(player.radius * player.sizeMult, Math.min(canvas.width - player.radius * player.sizeMult, player.x));
    player.y = Math.max(player.radius * player.sizeMult, Math.min(canvas.height - player.radius * player.sizeMult, player.y));
  }

  // spawn hazards
  spawnTimer -= dt;
  if (spawnTimer <= 0) {
    // spawn a hazard on left side, random y not too close to player start if within safe radius
    const y = Math.random() * canvas.height;
    const x = -HAZARD_RADIUS;
    hazards.push({ x, y, vx: HAZARD_SPEED, vy: 0, radius: HAZARD_RADIUS });
    spawnTimer = SPAWN_INTERVAL;
  }

  // update hazards
  for (let i = hazards.length - 1; i >= 0; i--) {
    const h = hazards[i];
    h.x += h.vx * dt;
    // remove off-screen right
    if (h.x - h.radius > canvas.width) {
      hazards.splice(i, 1);
      score++;
      beep(660, 0.03);
      continue;
    }

    // collision
    const dx = h.x - player.x;
    const dy = h.y - player.y;
    const distSq = dx * dx + dy * dy;
    const radSum = (player.radius * player.sizeMult) + h.radius;
    if (distSq < radSum * radSum) {
      phase = "gameover";
      beep(110, 0.25);
      // no break, allow other hazards removal?
    }
  }
}
