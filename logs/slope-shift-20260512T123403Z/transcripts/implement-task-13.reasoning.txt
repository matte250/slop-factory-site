function update(dt) {
  if (phase !== 'playing') return;

  // Player movement
  player.y += PLAYER_SPEED_Y * dt;
  player.x += tilt * PLAYER_SPEED_X * dt;

  // Keep player within canvas bounds horizontally; if out of bounds => game over.
  if (player.x < 0 || player.x > canvas.width) {
    phase = 'gameover';
    // maybe beep
    return;
  }

  // Hazard spawn timer
  hazardSpawnTimer -= dt;
  if (hazardSpawnTimer <= 0) {
    // spawn new hazard
    const side = Math.random() < 0.5 ? 'left' : 'right'; // which side is open gap?
    // we define hazard with y = player.y + SAFE_RADIUS + 200 or something.
    // But to ensure safe distance, we choose y = player.y + 400
    const spawnY = player.y + 400;
    hazards.push({ y: spawnY, side });
    hazardSpawnTimer = HAZARD_SPAWN_INTERVAL;
  }

  // Check hazards that have been passed (player.y >= hazard.y)
  for (let i = hazards.length - 1; i >= 0; i--) {
    const h = hazards[i];
    if (player.y >= h.y) {
      // determine if player is on correct side
      const onLeftSide = player.x < canvas.width / 2;
      const correct = (h.side === 'left' && onLeftSide) || (h.side === 'right' && !onLeftSide);

      if (!correct) {
        // player collided
        phase = 'gameover';
      } else {
        // survived hazard
        score++;
      }
      // remove hazard
      hazards.splice(i, 1);
    }
  }

  // Optional: If player goes too far down (score handling etc.)
  // Continue playing until lose.

  // Can also implement win condition after certain score e.g., score >= 10, set phase='won'
  if (score >= 10 && phase === 'playing') {

// === UPDATE ==============================================================
function update(dt) {
  if (phase !== "playing") return;

  // Player movement based on tilt
  player.y += PLAYER_SPEED_Y * dt;
  player.x += tilt * PLAYER_SPEED_X * dt;

  // Keep player within horizontal bounds; if out of bounds, game over.
  if (player.x < 0 || player.x > canvas.width) {
    phase = "gameover";
    return;
  }

  // Spawn hazards
  hazardSpawnTimer -= dt;
  if (hazardSpawnTimer <= 0) {
    // Ensure safe spawn distance: hazard y ahead of player
    const side = Math.random() < 0.5 ? "left" : "right";
    const spawnY = player.y + 400; // spawn ahead
    hazards.push({ y: spawnY, side });
    hazardSpawnTimer = HAZARD_SPAWN_INTERVAL;
  }

  // Check hazards
  for (let i = hazards.length - 1; i >= 0; i--) {
    const h = hazards[i];
    if (player.y >= h.y) {
      // Player has reached hazard line
      const onLeftSide = player.x < canvas.width / 2;
      const survived = (h.side === "left" && onLeftSide) || (h.side === "right" && !onLeftSide);
      if (!survived) {
        phase = "gameover";
      } else {
        score++;
      }
      hazards.splice(i, 1);
    }
  }

  // Optionally win condition after some score
  if (score >= 20 && phase === "playing") {
    phase = "won";
  }
}

function beep(freq, durSec) {
  try {
    if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = "square";
    osc.frequency.value = freq;
    gain.gain.setValueAtTime(0.05, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + durSec);
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + durSec);

}
