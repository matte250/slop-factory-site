```js
function update(dt) {
  if (phase !== "playing") return;
  // Move player
  player.x += tiltDirection * PLAYER_SPEED_X * dt;
  player.y += PLAYER_SPEED_Y * dt;
  // clamp horizontal
  player.x = clamp(player.x, player.radius, canvas.width - player.radius);
  // Remove hazards that are off screen
  // Update hazards
  for (const h of hazards) {
    h.y += PLAYER_SPEED_Y * dt; // move hazards downward
  }
  // collision detection
  for (let i = hazards.length - 1; i >= 0; i--) {
    const h = hazards[i];
    // Check if hazard passed player
    if (h.y > player.y + player.radius) {
      // Passed safely, increment score
      score++;
      hazards.splice(i, 1);
      continue;
    }
    // Check collision (player within hazard vertical range)
    if (h.y + HAZARD_HEIGHT >= player.y - player.radius && h.y <= player.y + player.radius) {
      // Determine if player is in gap
      const inGap = (h.side === "left" && player.x <= GAP_SIZE) ||
                    (h.side === "right" && player.x >= canvas.width - GAP_SIZE);
      if (!inGap) {
        phase = "gameover";
        beep(110, 0.3);
        // Cancel loop after this frame
        // We'll stop after rendering
      }
    }
  }
  // Spawn hazards
  spawnTimer -= dt;
  if (spawnTimer <= 0) {
    // toggle side
    hazardSide = hazardSide === "left" ? "right" : "left";
    hazards.push({ y: -HAZARD_HEIGHT, side: hazardSide });
    spawnTimer = HAZARD_SPAWN_INTERVAL;
  }
}
```
