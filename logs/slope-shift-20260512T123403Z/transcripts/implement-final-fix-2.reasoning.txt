function startLoop() {
  if (loopId !== null) cancelAnimationFrame(loopId);
  function tick(now) {
    try {
      const dt = Math.min((now - lastTime) / 1000, 0.1);
      lastTime = now;
      // Update game state.
      update(dt);
      // Render frame.
      render();
      // Publish observable.
      publishState();
    } catch (err) {
      console.error("loop error", err);
    }
    loopId = requestAnimationFrame(tick);
  }

}

function update(dt) {
  if (phase !== "playing") return;

  // move player
  player.x += tiltDirection * PLAYER_SPEED_X * dt;
  player.y += PLAYER_SPEED_Y * dt;

  // keep within horizontal bounds
  if (player.x < player.radius) {
    player.x = player.radius;
    // bounce off or clamp
  } else if (player.x > canvas.width - player.radius) {
    player.x = canvas.width - player.radius;
  }

  // simple death condition: fell off bottom
  if (player.y - player.radius > canvas.height) {
    phase = "gameover";
    // stop loop in next tick? keep it but after gameover we could cancel loop.
    if (loopId !== null) cancelAnimationFrame(loopId);
    loopId = null;
    return;
  }

  // score based on vertical progress
  score = Math.floor(player.y); // or increment per dt
}

function beep(freq, durSec) {
  try {
    if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = "square";
    osc.frequency.value = freq;
    gain.gain.setValueAtTime(0.05, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + durSec);
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + durSec);
  } catch {}
}
