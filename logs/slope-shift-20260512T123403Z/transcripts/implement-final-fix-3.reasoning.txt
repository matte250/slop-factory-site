function update(dt) {
  if (phase !== "playing") return;
  // Player movement
  player.x += tiltDirection * PLAYER_SPEED_X * dt;
  // clamp x within bounds
  player.x = Math.max(player.radius, Math.min(canvas.width - player.radius, player.x));
  
  // spawn hazards
  spawnTimer -= dt;
  if (spawnTimer <= 0) {
    // spawn new hazard at top
    hazards.push({ y: -HAZARD_HEIGHT, side: nextHazardSide });
    nextHazardSide = -nextHazardSide;
    spawnTimer = HAZARD_SPAWN_INTERVAL;
  }

  // Move hazards down
  for (let h of hazards) {
    h.y += PLAYER_SPEED_Y * dt;
  }

  // Collision detection
  for (let i = hazards.length - 1; i >= 0; i--) {
    const h = hazards[i];
    // Remove off-screen hazards and increase score
    if (h.y > canvas.height) {
      hazards.splice(i, 1);
      score++;
      continue;
    }
    // Check collision if overlapping vertically
    if (player.y + player.radius > h.y && player.y - player.radius < h.y + HAZARD_HEIGHT) {
      // Determine blocked region
      let blockedX, blockedW;
      if (h.side === 1) {
        // gap left, block right
        blockedX = GAP_SIZE;
        blockedW = canvas.width - GAP_SIZE;
      } else {
        // gap right, block left
        blockedX = 0;
        blockedW = canvas.width - GAP_SIZE;
      }
      if (player.x + player.radius > blockedX && player.x - player.radius < blockedX + blockedW) {
        // collision
        phase = "gameover";
        beep(110, 0.25);
        break;
      }
    }
  }
}
```

function render() {
  // Background
  ctx.fillStyle = "#0f1320";
  ctx.fillRect(0, 0, canvas.width, canvas.height);
  // Hazards
  ctx.fillStyle = "#7df9ff";
  for (const h of hazards) {
    if (h.side === 1) {
      // block right
      ctx.fillRect(GAP_SIZE, h.y, canvas.width - GAP_SIZE, HAZARD_HEIGHT);
    } else {
      // block left
      ctx.fillRect(0, h.y, canvas.width - GAP_SIZE, HAZARD_HEIGHT);
    }
  }
  // Player
  ctx.fillStyle = "#ff00aa";

// === AUDIO (lazy) =======================================================
function beep(freq, durSec) {
  try {
    if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = "square";
    osc.frequency.value = freq;
    gain.gain.setValueAtTime(0.05, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + durSec);
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + durSec);
  } catch {}
}
```

// === LOOP ==============================================================
function startLoop() {
  if (loopId !== null) cancelAnimationFrame(loopId);
  function tick(now) {
    try {
      const dt = Math.min((now - lastTime) / 1000, 0.1);
      lastTime = now;
      ctx.clearRect(0, 0, canvas.width, canvas.height);
      if (phase === "playing") {
        update(dt);
        render();
        publishState();
      } else {
        render(); // render gameover overlay
        publishState();
      }
    } catch (err) {
      console.error("loop error", err);
    }
    loopId = requestAnimationFrame(tick);
  }
  loopId = requestAnimationFrame(tick);
}
