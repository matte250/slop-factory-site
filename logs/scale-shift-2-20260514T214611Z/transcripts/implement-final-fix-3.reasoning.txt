function update(dt) {
   // player movement
   const moveSpeed = 400; // px/sec
   let dir = 0;
   if (keys.has('ArrowLeft') || keys.has('a') || keys.has('A')) dir -= 1;
   if (keys.has('ArrowRight') || keys.has('d') || keys.has('D')) dir += 1;
   player.x += dir * moveSpeed * dt;
   // clamp
   player.x = Math.max(playerRadius, Math.min(canvas.width - playerRadius, player.x));
   // update player radius based on size
   playerRadius = playerSize === 'large' ? PLAYER_LARGE_R : PLAYER_SMALL_R;

   // spawn
   spawnTimer -= dt;
   if (spawnTimer <= 0) {
       spawnTimer = SPAWN_INTERVAL + Math.random()*1; // random 
       // decide obstacle type
       const type = Math.random() < 0.5 ? 'low' : 'narrow';
       let w, h;
       if (type === 'low') {
           w = 100;
           h = 20;
       } else {
           w = 20;
           h = 100;
       }
       const x = Math.random() * (canvas.width - w);
       const y = -h;
       hazards.push({ x, y, w, h, type, alive: true });
   }

   // update hazards position
   for (const haz of hazards) {
       haz.y += HAZARD_SPEED * dt;
   }
   // remove off-screen hazards
   hazards = hazards.filter(h => h.y < canvas.height && h.alive);

   // check collisions
   for (const haz of hazards) {
       if (!haz.alive) continue;
       // simple AABB collision with circle
       const dx = Math.max(haz.x, Math.min(player.x, haz.x + haz.w));
       const dy = Math.max(haz.y, Math.min(player.y, haz.y + haz.h));
       const distSq = (dx - player.x) ** 2 + (dy - player.y) ** 2;
       if (distSq < playerRadius ** 2) {
           // collision
           if (haz.type === 'low') {
               if (playerSize === 'large') {
                   // smash
                   haz.alive = false;
                   score += 10;
                   beep(880, 0.05);
               } else {
                   // hit
                   phase = 'gameover';
                   beep(110, 0.25);
               }
           } else { // narrow

                   // slip through: just do nothing, maybe score
                   // skip: allow to pass

                   // hit
                   phase = 'gameover';
                   beep(110, 0.25);
               }
           }
           if (phase === 'gameover') {
               // break out
               break;
           }
       }
   }

   // if gameover, stop loop maybe set a flag and cancel?
   if (phase === 'gameover') {
       // set state and stop?

      // Player movement
      const dir = (keys.has('ArrowRight') || keys.has('d') || keys.has('D') ? 1 : 0)
                - (keys.has('ArrowLeft') || keys.has('a') || keys.has('A') ? 1 : 0);
      const radius = playerSize === 'large' ? PLAYER_LARGE_R : PLAYER_SMALL_R;
      player.x += dir * PLAYER_SPEED * dt;
      // Clamp within canvas bounds
      player.x = Math.max(radius, Math.min(canvas.width - radius, player.x));
      // Spawn hazards
      spawnTimer -= dt;
      if (spawnTimer <= 0) {
        spawnTimer = SPAWN_INTERVAL + Math.random() * 1;
        const type = Math.random() < 0.5 ? 'low' : 'narrow';
        let w, h;
        if (type === 'low') {
          w = 100;
          h = 20;
        } else {
          w = 20;
          h = 100;
        }
        const x = Math.random() * (canvas.width - w);
        const y = -h;
        hazards.push({ x, y, w, h, type, alive: true });
      }
      // Update hazards
      for (const hz of hazards) {
        hz.y += HAZARD_SPEED * dt;
      }
      // Remove off-screen hazards
      hazards = hazards.filter(hz => hz.y < canvas.height && hz.alive);
      // Collision detection
      const currentRadius = playerSize === 'large' ? PLAYER_LARGE_R : PLAYER_SMALL_R;
      for (const hz of hazards) {
        if (!hz.alive) continue;
        // Simple circle-rect collision
        const nearestX = Math.max(hz.x, Math.min(player.x, hz.x + hz.w));
        const nearestY = Math.max(hz.y, Math.min(player.y, hz.y + hz.h));
        const dx = player.x - nearestX;
        const dy = player.y - nearestY;
        const distSq = dx * dx + dy * dy;
        if (distSq < currentRadius * currentRadius) {
          if (hz.type === 'low') {
            if (playerSize === 'large') {
              hz.alive = false;
              score += 10;
              beep(880, 0.05);
            } else {
              phase = 'gameover';
              beep(110, 0.25);
            }
          } else { // narrow
            if (playerSize === 'small') {
              // successfully slip, score
              // score on passing will be handled when hazard passes player
            } else {
              phase = 'gameover';
              beep(110, 0.25);
            }
          }
          break;
        }
      }
    }
