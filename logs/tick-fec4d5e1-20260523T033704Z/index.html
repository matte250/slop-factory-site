        gameOver = true;

    }

  function draw() {

      if (s.y > canvas.height) s.y = 0;

    // draw lanes (optional visual aid)
    ctx.strokeStyle = '#222';
    for (let i = 1; i < LANE_COUNT; i++) {
      ctx.beginPath();
      ctx.moveTo(i * LANE_WIDTH, 0);
      ctx.lineTo(i * LANE_WIDTH, canvas.height);

    const delta = time - lastTime;
