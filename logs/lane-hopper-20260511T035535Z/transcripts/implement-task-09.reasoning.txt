  // background
  ctx.fillStyle = "#0f1320";
  ctx.fillRect(0, 0, canvas.width, canvas.height);

  // lanes

    const y = i * LANE_HEIGHT;
    ctx.beginPath();
    ctx.moveTo(0, y);
    ctx.lineTo(canvas.width, y);

  // hazards
  ctx.fillStyle = "#ff52ae";
  hazards.forEach(h => {
    const y = h.lane * LANE_HEIGHT + LANE_HEIGHT / 2;
    ctx.beginPath();
