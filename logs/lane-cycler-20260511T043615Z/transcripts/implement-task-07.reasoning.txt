  // === UPDATE ==============================================================
  function update(dt) {
    // Update player Y based on lane
    player.y = player.lane * LANE_HEIGHT + LANE_HEIGHT / 2;

    // Update spawn timer
    spawnTimer -= dt;
    if (spawnTimer <= 0) {
      spawnHazard();
      spawnTimer = MIN_SPAWN_INTERVAL + Math.random() * (MAX_SPAWN_INTERVAL - MIN_SPAWN_INTERVAL);
    }

    // Move hazards
    for (let i = hazards.length - 1; i >= 0; i--) {
      const h = hazards[i];
      h.x -= hazardSpeed * dt;
      if (h.x + h.width < 0) {
        hazards.splice(i, 1);
        score++; // Increment score for each passed hazard
      }
    }

    // Collision detection
    for (const h of hazards) {
      if (h.x <= player.x && h.x + h.width >= player.x && player.lane !== h.gapLane) {
        gameOver();
        break;
      }
    }
  }

  // === RENDER ==============================================================
  function render() {
    // Clear canvas
    ctx.fillStyle = '#0f1320';
    ctx.fillRect(0, 0, CANVAS_WIDTH, CANVAS_HEIGHT);

    // Draw hazards
    ctx.fillStyle = '#ff0055';
    for (const h of hazards) {
      // Top solid part
      if (h.gapLane > 0) {
        ctx.fillRect(h.x, 0, h.width, h.gapLane * LANE_HEIGHT);
      }
      // Bottom solid part
      const bottomY = (h.gapLane + 1) * LANE_HEIGHT;
      const bottomHeight = CANVAS_HEIGHT - bottomY;
      if (bottomHeight > 0) {
        ctx.fillRect(h.x, bottomY, h.width, bottomHeight);
      }
    }

    // Draw player
    ctx.fillStyle = '#7df9ff';
    ctx.beginPath();
    ctx.arc(player.x, player.y, 8, 0, Math.PI * 2);
    ctx.fill();

    // Draw HUD - score
    ctx.fillStyle = '#e6e6f0';
    ctx.font = '16px ui-sans-serif, system-ui';
    ctx.textAlign = 'left';
    ctx.fillText('Score ' + score, 12, 22);
