```js
const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');
canvas.width = window.innerWidth;
canvas.height = window.innerHeight;

let orb = {x: canvas.width/2, y: canvas.height/2, r:10, vx:0, vy:2};
let obstacles = [];
let speed = 2;
let score = 0;
let lastObstacle = 0;

function resize(){ canvas.width = window.innerWidth; canvas.height = window.innerHeight; }
window.addEventListener('resize', resize);

function addObstacle(){
  const size = 20+Math.random()*30;
  const side = Math.floor(Math.random()*4);
  let x,y,dx,dy;
  switch(side){
    case 0: x = -size; y = Math.random()*canvas.height; dx = speed; dy = 0; break;
    case 1: x = canvas.width+size; y = Math.random()*canvas.height; dx = -speed; dy =0; break;
    case 2: x = Math.random()*canvas.width; y = -size; dx =0; dy = speed; break;
    case 3: x = Math.random()*canvas.width; y = canvas.height+size; dx =0; dy = -speed; break;
  }
  obstacles.push({x,y,r:size/2,dx,dy});
}

function update(dt){
  // move orb forward automatically
  orb.x += orb.vx*dt;
  orb.y += orb.vy*dt;
  // wrap around edges? lose if out of bounds
  if(orb.x<0||orb.x>canvas.width||orb.y<0||orb.y>canvas.height){ gameOver();}
  // obstacles move
  obstacles.forEach(o=>{o.x+=o.dx*dt; o.y+=o.dy*dt;});
  // remove offscreen obstacles
  obstacles = obstacles.filter(o=> o.x>-50 && o.x<canvas.width+50 && o.y>-50 && o.y<canvas.height+50);
  // collision
  for (let o of obstacles){
    const dx=o.x-orb.x, dy=o.y-orb.y;
    if (Math.hypot(dx,dy) < o.r+orb.r){ gameOver();}
  }
  // spawn
  if (performance.now()-lastObstacle>1500){
    addObstacle();
    lastObstacle = performance.now();
  }
  // increase speed slowly
  speed += 0.00001;
  orb.vx = 0; orb.vy = speed;
  score += dt*0.01;
}

let last = performance.now();
let running = true;
function loop(){
  if(!running) return;
  const now = performance.now();
  const dt = (now-last)/16; // normalize ~60fps
  last = now;
  update(dt);
  draw();
  requestAnimationFrame(loop);
}
function draw(){
  ctx.fillStyle='#111';
  ctx.fillRect(0,0,canvas.width,canvas.height);
  // orb
  ctx.beginPath();
  ctx.arc(orb.x,orb.y,orb.r,0,2*Math.PI);
  ctx.fillStyle='cyan';
  ctx.fill();
  // obstacles
  ctx.fillStyle='magenta';
  for(let o of obstacles){
    ctx.beginPath();
    ctx.arc(o.x,o.y,o.r,0,2*Math.PI);
    ctx.fill();
  }
  // score
  ctx.fillStyle='white';
  ctx.font='20px sans-serif';
  ctx.fillText('Score: '+Math.floor(score),10,30);
}
function gameOver(){
  running = false;
  ctx.fillStyle='red';
  ctx.font='40px sans-serif';
  ctx.fillText('Game Over', canvas.width/2-100, canvas.height/2);
}
canvas.addEventListener('click',()=>{ // toggle direction
  // rotate velocity clockwise 90deg
  const nx = orb.vy;
  const ny = -orb.vx;
  orb.vx = nx;
  orb.vy = ny;
});
loop();
```
