  function draw() {
    // Background starfield
    ctx.fillStyle = '#000010';
    ctx.fillRect(0, 0, width, height);
    stars.forEach(s => {
      ctx.fillStyle = 'white';
      ctx.fillRect(s.x, s.y, 1, 1);
    });

    // Ship – rotated triangle with gradient
    const grad = ctx.createLinearGradient(0, 0, 0, ship.size * 2);
    grad.addColorStop(0, '#00f');
    grad.addColorStop(1, '#0ff');
    ctx.fillStyle = grad;
    ctx.save();
    ctx.translate(ship.x, ship.y);
    ctx.rotate(ship.angle);
    ctx.beginPath();
    ctx.moveTo(0, -ship.size);
    ctx.lineTo(-ship.size, ship.size);
    ctx.lineTo(ship.size, ship.size);
    ctx.closePath();
    ctx.fill();
    ctx.restore();

    // Thruster flame when moving
    if (ship.dx !== 0 || ship.dy !== 0) {
      ctx.fillStyle = 'orange';
      ctx.beginPath();
      ctx.moveTo(0, ship.size);
      ctx.lineTo(-ship.size / 2, ship.size + 6);
      ctx.lineTo(ship.size / 2, ship.size + 6);
      ctx.closePath();
      ctx.save();
      ctx.translate(ship.x, ship.y);
      ctx.rotate(ship.angle);
      ctx.fill();
      ctx.restore();
    }

    // Asteroids – circles with gradient
    asteroids.forEach(a => {
      const radial = ctx.createRadialGradient(a.x, a.y, a.r * 0.3, a.x, a.y, a.r);
      radial.addColorStop(0, '#888');
      radial.addColorStop(1, '#444');
      ctx.fillStyle = radial;
      ctx.beginPath();
      ctx.arc(a.x, a.y, a.r, 0, Math.PI * 2);
      ctx.fill();
    });

    // Score
    ctx.fillStyle = 'white';
    ctx.font = '16px sans-serif';
    ctx.fillText('Score: ' + score, 10, 20);
    // Game over overlay
    if (gameOver) {
      ctx.fillStyle = 'rgba(0,0,0,0.6)';
      ctx.fillRect(0, 0, width, height);
      ctx.fillStyle = 'red';
      ctx.font = '32px sans-serif';
      ctx.textAlign = 'center';
      ctx.fillText('Game Over', width / 2, height / 2);
    }
  }
