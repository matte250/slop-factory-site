# Game Idea: Canvas Escape

**Concept**: A top‑down view where the player controls a tiny square navigating a maze of moving barriers inside an HTML canvas. The goal is to collect glowing tokens while avoiding colliding with any barrier.

**Lose Condition**: The game ends when the player collides with a barrier or fails to collect a token within a time limit.

**Core Mechanics**:
- Arrow keys or WASD to move the square.
- Barriers move in predictable patterns.
- Tokens appear randomly and grant points.
- Simple graphics and physics suitable for plain Canvas API.

**Why It Works**: Requires only basic drawing, collision detection, and input handling—perfect for an HTML canvas prototype.

  const barriers = [];
  const tokens = [];
  let lastTokenTime = 0;
  let score = 0;
  let gameOver = false;

  const rectIntersect = (a, b) =>

    a.y < b.y + b.size && a.y + a.size > b.y;

    if (!audioStarted) {

      audioStarted = true;

    const gain = audioCtx.createGain();

    osc.type = 'sine';
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    gain.gain.setValueAtTime(0.001, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.2, audioCtx.currentTime + 0.01);
    osc.start();
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + duration);
    osc.stop(audioCtx.currentTime + duration);
  };

  const playCollisionSound = () => playTone(200, 0.3);

  // Input handling
  window.addEventListener('keydown', e => { keys[e.key] = true; ensureAudio(); });
  window.addEventListener('keyup', e => { keys[e.key] = false; });
  window.addEventListener('keydown', e => { keys[e.key] = true; });
  window.addEventListener('keyup', e => { keys[e.key] = false; });

      const barrier = {
        x: horiz ? Math.random() * (width - 100) : Math.random() * width,
        y: horiz ? Math.random() * height : Math.random() * (height - 100),

        h: horiz ? 20 : 100,
        dx: horiz ? (Math.random() > 0.5 ? 1 : -1) * 1 : 0,
        dy: horiz ? 0 : (Math.random() > 0.5 ? 1 : -1) * 1,
        size: 0 // unused; keep API same as player/token
      };
      barriers.push(barrier);
    }
  };

  // Spawn a token at a random location
  const spawnToken = () => {
    const token = {
      x: Math.random() * (width - 20),
      y: Math.random() * (height - 20),
      size: 20,
      glow: Math.random() * 0.5 + 0.5 // for visual effect
    };
    tokens.push(token);
  };

  // Update loop
  const update = dt => {
    if (gameOver) return;
    // Move player
    if (keys['ArrowUp'] || keys['w']) player.y -= player.speed;
    if (keys['ArrowDown'] || keys['s']) player.y += player.speed;
    if (keys['ArrowLeft'] || keys['a']) player.x -= player.speed;
    if (keys['ArrowRight'] || keys['d']) player.x += player.speed;
    // Clamp
    player.x = Math.max(0, Math.min(width - player.size, player.x));
    player.y = Math.max(0, Math.min(height - player.size, player.y));

      b.y += b.dy;

        player.y < b.y + b.h && player.y + player.size > b.y

        playCollisionSound();
          gameOver = true;

          tokens.splice(i, 1);
          score++;

      lastTokenTime = Date.now();

    ctx.lineTo(x, y + radius);
    ctx.quadraticCurveTo(x, y, x + radius, y);
    ctx.closePath();
    ctx.fill();
    ctx.restore();

      token.y + token.size / 2,
      token.size * 0.2,

      token.y + token.size / 2,

    ctx.fill();
    ctx.restore();
  };

    ctx.fillStyle = '#111';
    ctx.fillRect(0, 0, width, height);
    // Player with gradient rounded rect
    const playerGrad = ctx.createLinearGradient(0, 0, player.size, player.size);
    playerGrad.addColorStop(0, '#4a90e2');
    playerGrad.addColorStop(1, '#007aff');
    drawRoundedRect(player.x, player.y, player.size, player.size, 4, playerGrad);
    // Barriers with red gradient

    barrierGrad.addColorStop(0, '#ff4d4d');

    ctx.fillStyle = '#00f';
    ctx.fillRect(player.x, player.y, player.size, player.size);
    // Barriers
    ctx.fillStyle = '#f00';
    barriers.forEach(b => ctx.fillRect(b.x, b.y, b.w, b.h));

      ctx.fillRect(t.x, t.y, t.size, t.size);
