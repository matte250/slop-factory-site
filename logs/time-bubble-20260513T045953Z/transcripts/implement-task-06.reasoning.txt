```js
function spawnHazard() {
    // spawn hazard with random position not within safe radius of player
    const maxAttempts = 30;
    let x, y;
    for (let i = 0; i < maxAttempts; i++) {
        x = Math.random() * W;
        y = Math.random() * H;
        if (Math.hypot(x - player.x, y - player.y) >= SAFE_RADIUS) break;
    }
    const speed = 100 + Math.random() * 100; // arbitrary
    const angle = Math.random() * Math.PI * 2;
    hazards.push({
        x, y,
        vx: Math.cos(angle) * speed,
        vy: Math.sin(angle) * speed,
        r: 12,
    });
}
function spawnHazards(n) {
    for (let i = 0; i < n; i++) spawnHazard();
}
```
