function beep(freq, durSec) {
  try {
    if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = 'sine';
    osc.frequency.value = freq;

// === UPDATE =============================================================
function update(dt) {
  if (phase !== 'playing') return;
  // Player movement
  const thrust = PLAYER_THRUST;
  if (keys.has('ArrowLeft') || keys.has('a') || keys.has('A')) player.vx -= thrust * dt;
  if (keys.has('ArrowRight') || keys.has('d') || keys.has('D')) player.vx += thrust * dt;
  if (keys.has('ArrowUp') || keys.has('w') || keys.has('W')) player.vy -= thrust * dt;
  if (keys.has('ArrowDown') || keys.has('s') || keys.has('S')) player.vy += thrust * dt;
  // Apply friction
  const friction = Math.pow(PLAYER_FRICTION, dt);
  player.vx *= friction;
  player.vy *= friction;
  // Clamp speed
  const speed = Math.hypot(player.vx, player.vy);
  if (speed > PLAYER_MAX_SPEED) {
    player.vx = (player.vx / speed) * PLAYER_MAX_SPEED;
    player.vy = (player.vy / speed) * PLAYER_MAX_SPEED;
  }
  // Update position
  player.x = Math.min(Math.max(player.x + player.vx * dt, 0), W);
  player.y = Math.min(Math.max(player.y + player.vy * dt, 0), H);

  // Bubble timer
  if (bubbleActive) {
    bubbleTimer += dt;
    if (bubbleTimer >= BUBBLE_DURATION) {
      bubbleActive = false;
      bubbleTimer = 0;
    }
  }

  // Hazard spawning
  hazardSpawnTimer -= dt;
  if (hazardSpawnTimer <= 0) {
    spawnHazard();
    hazardSpawnTimer = 1.0 + Math.random() * 1.0;
  }

  // Update hazards
  for (let i = hazards.length - 1; i >= 0; i--) {
    const h = hazards[i];
    if (!bubbleActive) {
      h.x += h.vx * dt;
      h.y += h.vy * dt;
    }
    // Remove and score if passed bottom
    if (h.y - h.r > H) {
      hazards.splice(i, 1);
      score += 1;
      beep(600, 0.05);
      continue;
    }
    // Collision with player

    }
  }
