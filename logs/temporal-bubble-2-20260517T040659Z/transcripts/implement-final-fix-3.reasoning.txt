function spawnHazard() {
  // spawn at random edge, moving inward
  const edge = Math.random();
  let x, y, vx, vy;
  if (edge < 0.5) {
    // left or right
    const fromLeft = Math.random() < 0.5;
    x = fromLeft ? -HAZARD_RADIUS : W + HAZARD_RADIUS;
    y = Math.random() * H;
    const speed = HAZARD_SPEED_MIN + Math.random() * (HAZARD_SPEED_MAX - HAZARD_SPEED_MIN);
    vx = fromLeft ? speed : -speed;
    vy = (Math.random() - 0.5) * speed * 0.5;
  } else {
    // top or bottom
    const fromTop = Math.random() < 0.5;
    x = Math.random() * W;
    y = fromTop ? -HAZARD_RADIUS : H + HAZARD_RADIUS;
    const speed = HAZARD_SPEED_MIN + Math.random() * (HAZARD_SPEED_MAX - HAZARD_SPEED_MIN);
    vy = fromTop ? speed : -speed;
    vx = (Math.random() - 0.5) * speed * 0.5;
  }
  const r = HAZARD_RADIUS;
  // Ensure safe radius from player at spawn
  const dx = x - player.x;
  const dy = y - player.y;
  const dist = Math.hypot(dx, dy);
  if (dist < SAFE_RADIUS_AT_SPAWN) {
    // don't spawn this hazard now; just skip
    return;
  }
  hazards.push({ x, y, vx, vy, r });
}
