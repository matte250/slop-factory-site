    function update(dt) {
      if (phase !== 'playing') return;
      // Spawn hazards
      spawnTimer -= dt;
      if (spawnTimer <= 0) {
        spawnHazard();
        spawnTimer = SPAWN_INTERVAL;
      }
      // Update hazards
      for (const h of hazards) {
        h.y += OBSTACLE_SPEED * dt;
        h.angle += h.spinDir * SPIN_SPEED * dt;
      }
      // Remove hazards that passed beyond canvas and increase score
      for (let i = hazards.length - 1; i >= 0; i--) {
        const h = hazards[i];
        if (h.y - h.r > H) {
          // hazard passed player without hitting
          score++;
          hazards.splice(i, 1);
        }
      }
      // Collision detection
      for (const h of hazards) {
        const dx = player.x - h.x;
        const dy = player.y - h.y;
        const dist = Math.hypot(dx, dy);
        if (dist < h.r) {
          // Determine angle to player relative to hazard rotation
          const angleToPlayer = Math.atan2(dy, dx);
          let diff = (angleToPlayer - h.angle) % (2 * Math.PI);
          if (diff < 0) diff += 2 * Math.PI;
          // if not within gap, collision
          if (diff > GAP_ANGLE && diff < 2 * Math.PI - GAP_ANGLE) {
            phase = 'gameover';
            beep(200, 0.4);
            break;
          }
        }
      }

function beep(freq, durSec) {
  try {
    if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = 'triangle';
    osc.frequency.value = freq;
    gain.gain.setValueAtTime(0.04, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + durSec);
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + durSec);
  } catch (e) {
    // fail silently
  }
}
