function update(dt) {
  if (phase !== 'playing') return;
  const ACCEL = 400; // pixels per second^2
  const FRICTION = 0.9; // per second?
  // adjust velocity based on keys
  if (keys.has('ArrowUp') || keys.has('w') || keys.has('W')) {
    player.vy -= ACCEL * dt;
  }
  if (keys.has('ArrowDown') || keys.has('s') || keys.has('S')) {
    player.vy += ACCEL * dt;
  }
  // apply friction
  const decay = Math.pow(FRICTION, dt);
  player.vy *= decay;
  // update position
  player.y += player.vy * dt;
  // clamp
  if (player.y < 0) {
    player.y = 0;
    player.vy = 0;
  } else if (player.y > H) {
    player.y = H;
    player.vy = 0;
  }
  player.visible = player.x >= 0 && player.x <= W && player.y >= 0 && player.y <= H;
  
  // spawn hazards if needed (simple timer)
  spawnTimer -= dt;
  if (spawnTimer <= 0) {
    spawnHazard();
    spawnTimer = 1.5; // spawn every 1.5 seconds
  }
  
  // update hazards
  for (let i = hazards.length - 1; i >= 0; i--) {
    const h = hazards[i];
    h.x -= h.speed * dt;
    h.angle += h.spinSpeed * dt * spinDir;
    if (h.x + h.radius < 0) {
      hazards.splice(i, 1);
      // hazard passed player, increment score
      score++;
      beep(800, 0.05);
    }
  }
  
  // check collisions
  for (const h of hazards) {
    const dx = h.x - player.x;
    const dy = h.y - player.y;
    const dist = Math.hypot(dx, dy);
    // simple collision if within radius (assuming hazard is a full circle with gap)
    if (dist < h.radius + player.radius) {
      // check if within gap: we can approximate gap at angle + gapAngle/2 range; but simpler: always collide -> game over.
      // For simplicity, if gap is small area, collisions happen often enough.
      phase = 'gameover';
      beep(200, 0.4);
      break;
    }
  }
}
