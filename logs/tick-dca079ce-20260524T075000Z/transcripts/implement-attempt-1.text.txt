Implemented **game.js** – a concise asteroid‑dodge game that targets the `<canvas id="game">` element.

```javascript
// game.js – simple asteroid‑dodge game
(() => {
  const canvas = document.getElementById('game')
  if (!canvas) return
  const ctx = canvas.getContext('2d')

  // Canvas size (fallback if not set in HTML/CSS)
  canvas.width = canvas.width || 800
  canvas.height = canvas.height || 600

  // Ship
  const ship = {
    w: 40,
    h: 20,
    x: canvas.width / 2 - 20,
    y: canvas.height - 30,
    speed: 5,
  }

  // Asteroids
  const asteroids = []
  const asteroidSize = 30
  let asteroidTimer = 0
  const asteroidInterval = 90 // frames
  let speedIncrease = 0.001 // acceleration per frame

  // Score
  let score = 0
  let gameOver = false

  // Input handling
  const keys = { ArrowLeft: false, ArrowRight: false }
  window.addEventListener('keydown', e => {
    if (e.key in keys) keys[e.key] = true
  })
  window.addEventListener('keyup', e => {
    if (e.key in keys) keys[e.key] = false
  })

  function spawnAsteroid() {
    const x = Math.random() * (canvas.width - asteroidSize)
    asteroids.push({ x, y: -asteroidSize, w: asteroidSize, h: asteroidSize, v: 2 })
  }

  function update() {
    // Ship movement
    if (keys.ArrowLeft) ship.x = Math.max(0, ship.x - ship.speed)
    if (keys.ArrowRight) ship.x = Math.min(canvas.width - ship.w, ship.x + ship.speed)

    // Asteroid spawn
    if (asteroidTimer++ >= asteroidInterval) {
      spawnAsteroid()
      asteroidTimer = 0
    }

    // Move asteroids
    asteroids.forEach(a => {
      a.v += speedIncrease
      a.y += a.v
    })

    // Remove off‑screen asteroids & increase score
    for (let i = asteroids.length - 1; i >= 0; i--) {
      const a = asteroids[i]
      if (a.y > canvas.height) {
        asteroids.splice(i, 1)
        score++
      } else if (rectsCollide(a, ship)) {
        gameOver = true
      }
    }
  }

  function rectsCollide(r1, r2) {
    return (
      r1.x < r2.x + r2.w &&
      r1.x + r1.w > r2.x &&
      r1.y < r2.y + r2.h &&
      r1.y + r1.h > r2.y
    )
  }

  function draw() {
    ctx.clearRect(0, 0, canvas.width, canvas.height)

    // Ship
    ctx.fillStyle = '#0af'
    ctx.fillRect(ship.x, ship.y, ship.w, ship.h)

    // Asteroids
    ctx.fillStyle = '#a33'
    asteroids.forEach(a => ctx.fillRect(a.x, a.y, a.w, a.h))

    // Score
    ctx.fillStyle = '#000'
    ctx.font = '16px sans-serif'
    ctx.fillText(`Score: ${score}`, 10, 20)

    if (gameOver) {
      ctx.fillStyle = 'rgba(0,0,0,0.6)'
      ctx.fillRect(0, 0, canvas.width, canvas.height)
      ctx.fillStyle = '#fff'
      ctx.textAlign = 'center'
      ctx.font = '24px sans-serif'
      ctx.fillText('Game Over', canvas.width / 2, canvas.height / 2)
      ctx.fillText(`Final Score: ${score}`, canvas.width / 2, canvas.height / 2 + 30)
    }
  }

  function loop() {
    if (!gameOver) {
      update()
      draw()
      requestAnimationFrame(loop)
    } else {
      draw() // final frame with game‑over overlay
    }
  }

  // Start the loop
  requestAnimationFrame(loop)
})()
```
