function reset() {
    player.angle = 0;
    player.radius = PLAYER_ORBIT_INNER;
    player.x = CENTER_X + Math.cos(player.angle) * player.radius;
    player.y = CENTER_Y + Math.sin(player.angle) * player.radius;

Now // === LOOP/UPDATE =========================================================

function startLoop() {
  if (loopId !== null) cancelAnimationFrame(loopId);
  function tick(now) {
    try {
      const dt = Math.min((now - lastTime) / 1000, 0.1);
      lastTime = now;
      update(dt);
      render();
      publishState();
    } catch (err) {
      console.error('loop error', err);
    }
    loopId = requestAnimationFrame(tick);
  }
  loopId = requestAnimationFrame(tick);
}

function update(dt) {
  if (phase !== "playing") return;
  // Handle toggle
  if (toggleRequested) {
    player.radius = (player.radius === PLAYER_ORBIT_INNER) ? PLAYER_ORBIT_OUTER : PLAYER_ORBIT_INNER;
    toggleRequested = false;
    beep(440, 0.1);
  }
  // Update angle
  player.angle += PLAYER_SPEED * dt;
  // wrap angle
  if (player.angle > Math.PI * 2) player.angle -= Math.PI * 2;
  // compute position
  player.x = CENTER_X + Math.cos(player.angle) * player.radius;
  player.y = CENTER_Y + Math.sin(player.angle) * player.radius;
  // Score on full rotation
  if (prevAngle > player.angle) {
    score += 1;
    beep(660, 0.05);
  }
  prevAngle = player.angle;
  // Spawn hazards
  nextHazardSpawn -= dt;
  if (nextHazardSpawn <= 0) {
    spawnHazard();
    nextHazardSpawn = HAZARD_SPAWN_INTERVAL;
  }
  // Update hazards (if moving, but static)
  // Collision detection
  for (let i = hazards.length - 1; i >= 0; i--) {
    const h = hazards[i];
    const hx = CENTER_X + Math.cos(h.angle) * h.radius;
    const hy = CENTER_Y + Math.sin(h.angle) * h.radius;
    const dist = Math.hypot(player.x - hx, player.y - hy);
    if (dist < PLAYER_RADIUS + HAZARD_RADIUS) {
      phase = "gameover";
      beep(220, 0.3);
      break;
    }
  }
  // Remove hazards that are behind player? Not necessary for simplicity.
}

function spawnHazard() {
  // Random radius inner/outer
  const radius = Math.random() < 0.5 ? PLAYER_ORBIT_INNER : PLAYER_ORBIT_OUTER;
  // Choose angle not too close to player's current angle
  let angle;
  let attempts = 0;
  do {
    angle = Math.random() * Math.PI * 2;
    const diff = Math.abs(angle - player.angle);
    const delta = Math.min(diff, Math.PI * 2 - diff);
    attempts++;
    if (attempts > 30) break;
  } while (Math.abs(angle - player.angle) < SAFE_ANGLE);
  hazards.push({ angle, radius });
}

function render() {
  // Already cleared in tick
  // Draw central point
  ctx.fillStyle = "#7df9ff";
  ctx.beginPath();
  ctx.arc(CENTER_X, CENTER_Y, 4, 0, Math.PI * 2);
  ctx.fill();
  // Draw hazards

  for (const h of hazards) {
    const hx = CENTER_X + Math.cos(h.angle) * h.radius;
    const hy = CENTER_Y + Math.sin(h.angle) * h.radius;
    ctx.beginPath();
    ctx.arc(hx, hy, HAZARD_RADIUS, 0, Math.PI * 2);
    ctx.fill();
  }
  // Draw player

  ctx.beginPath();
  ctx.arc(player.x, player.y, PLAYER_RADIUS, 0, Math.PI * 2);

function beep(freq, durSec) {
  try {
    if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = "triangle";
    osc.frequency.value = freq;
    gain.gain.setValueAtTime(0.04, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + durSec);
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + durSec);
  } catch {}
}
