function update(dt) {
   if (phase !== "playing") return;

   // Player forward motion
   player.y -= PLAYER_SPEED * dt;
   // Keep player within canvas bounds (optional)
   if (player.y < 0) player.y = 0;

   // Update spawn timer
   spawnTimer -= dt;
   if (spawnTimer <= 0) {
       spawnHazard();
       spawnTimer = SPAWN_INTERVAL;
   }

   // Update hazards
   for (let i = hazards.length - 1; i >= 0; i--) {
       const h = hazards[i];
       h.y += h.speed * dt;
       // Collision detection
       const dx = h.x - player.x;
       const dy = h.y - player.y;
       const dist = Math.hypot(dx, dy);
       const collisionDist = HAZARD_RADIUS + PLAYER_RADIUS;
       if (dist < collisionDist) {
           if (h.colorIndex !== player.colorIndex) {
               // Colors differ, game over
               phase = "gameover";
               // Optionally beep
           } else {
               // Colors match, safe pass, will be removed later when off-screen and score incremented.
           }
       }

       // Remove hazard if off-screen
       if (h.y - HAZARD_RADIUS > canvas.height) {
           // Passed player successfully; increment score if still playing
           if (phase === "playing") {
               score++;
           }
           hazards.splice(i, 1);
       }
   }

   // Publishing observable state
   publishState();
}
